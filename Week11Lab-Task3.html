tourism <- read_csv(
  "singapore_tourism_receipts.csv",
  show_col_types = FALSE
)

item_order <- c(
  "Sightseeing, Entertainment & Gaming",
  "Shopping",
  "Other Components",
  "Food & Beverage",
  "Accommodation"
)

tourism_plot <- tourism |>
  mutate(
    quarter = factor(
      quarter,
      levels = sort(unique(quarter))
    ),

    item = factor(
      item,
      levels = item_order
    ),

    # Convert million SGD to billion SGD
    value_billion = value / 1000,

    # Only label larger sections to prevent overcrowding
    value_label = if_else(
      value_billion >= 2.5,
      sprintf("%.3f", value_billion),
      NA_character_
    )
  )
quarter_totals <- tourism_plot |>
  group_by(quarter) |>
  summarise(
    total_billion = sum(value_billion),
    .groups = "drop"
  )
events <- tibble(
  quarter_text = c(
    "2020-Q2",
    "2022-Q2"
  ),

  annotation_position = c(
    8.5,
    11.5
  ),

  event_label = c(
    "COVID-19 Circuit Breaker\nand border restrictions",
    "Singapore reopened borders\nunder the Vaccinated Travel Framework"
  )
) |>
  mutate(
    quarter = factor(
      quarter_text,
      levels = levels(tourism_plot$quarter)
    )
  ) |>
  left_join(
    quarter_totals,
    by = "quarter"
  )

receipt_colours <- c(
  "Sightseeing, Entertainment & Gaming" = "#00496F",
  "Shopping"                             = "#0F85A0",
  "Other Components"                    = "#F0DA3E",
  "Food & Beverage"                     = "#ED8B00",
  "Accommodation"                       = "#DD4124"
)

font_used <- "serif"


tourism_receipts_plot <- ggplot(
  tourism_plot,
  aes(
    x = quarter,
    y = value_billion,
    fill = item
  )
) +

  # Stacked bars
  geom_col(
    width = 0.88,
    position = position_stack(reverse = TRUE)
  ) +

  # Value labels
  geom_text(
    aes(label = value_label),
    position = position_stack(
      vjust = 0.5,
      reverse = TRUE
    ),
    family = font_used,
    size = 4.5,
    colour = "black",
    na.rm = TRUE
  ) +

  # Arrows pointing from the event labels to the relevant bars
  geom_segment(
    data = events,
    aes(
      x = quarter,
      xend = quarter,
      y = annotation_position - 1.8,
      yend = total_billion + 0.25
    ),
    inherit.aes = FALSE,
    colour = "#555555",
    linewidth = 0.4,
    arrow = grid::arrow(
      length = grid::unit(0.09, "inches"),
      type = "closed"

    data = events,

Tourism receipts fell sharply during the COVID-19 period, with the cumulative total reaching only S$4,188 billion in 2020-Q2, compared with S$27,689 billion in 2019-Q4. Recovery became clearer after Singapore reopened its borders, rising from S$4,574 billion in 2022-Q2 to S$14,178 billion in 2022-Q4. By 2024-Q4, total receipts reached S$29,781 billion, exceeding the 2019-Q4 level. Other Components was the largest category in 2024-Q4 at S$9,104 billion.
